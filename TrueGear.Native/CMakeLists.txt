cmake_minimum_required(VERSION 3.24)
project(Titanfall2VR_TrueGear LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with x64 MSVC on Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /EHsc /utf-8 /permissive-)
include_directories(src vendor)
add_library(tg_events STATIC src/event_router.cpp src/vr_rumble_filter.cpp)
add_library(tg_transport STATIC src/truegear_transport.cpp src/truegear_player.cpp)
target_link_libraries(tg_transport PUBLIC winhttp crypt32)
add_library(Titanfall2VR_TrueGear SHARED src/plugin.cpp src/runtime.cpp src/vr_haptic_bridge.cpp src/vr_rumble_bridge.cpp src/vr_custom_haptic_bridge.cpp)
target_link_libraries(Titanfall2VR_TrueGear PRIVATE tg_events tg_transport bcrypt)
option(BUILD_TESTING "Build optional developer tests" OFF)
if(BUILD_TESTING)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/queue_test.cpp")
  message(FATAL_ERROR "This source package omits developer tests. Configure with BUILD_TESTING=OFF.")
endif()
enable_testing()
add_executable(queue_test tests/queue_test.cpp)
add_test(NAME queue COMMAND queue_test)
add_executable(event_router_test tests/event_router_test.cpp)
target_link_libraries(event_router_test PRIVATE tg_events)
add_test(NAME event_router COMMAND event_router_test)
add_executable(plugin_smoke tests/plugin_smoke.cpp)
add_test(NAME plugin_abi COMMAND plugin_smoke $<TARGET_FILE:Titanfall2VR_TrueGear>)
add_executable(plugin_bootstrap_test tests/plugin_bootstrap_test.cpp src/vr_haptic_bridge.cpp src/vr_rumble_bridge.cpp src/vr_custom_haptic_bridge.cpp)
target_link_libraries(plugin_bootstrap_test PRIVATE bcrypt tg_events)
add_test(NAME plugin_bootstrap COMMAND plugin_bootstrap_test)
add_executable(vr_hand_state_test tests/vr_hand_state_test.cpp)
add_test(NAME vr_hand_state COMMAND vr_hand_state_test)
add_executable(vr_slot_state_test tests/vr_slot_state_test.cpp)
add_test(NAME vr_slot_state COMMAND vr_slot_state_test)
add_executable(vr_grenade_slot_state_test tests/vr_grenade_slot_state_test.cpp)
add_test(NAME vr_grenade_slot_state COMMAND vr_grenade_slot_state_test)
add_executable(vr_ammo_pouch_state_test tests/vr_ammo_pouch_state_test.cpp)
add_test(NAME vr_ammo_pouch_state COMMAND vr_ammo_pouch_state_test)
add_executable(vr_haptic_state_test tests/vr_haptic_state_test.cpp)
add_test(NAME vr_haptic_state COMMAND vr_haptic_state_test)
add_executable(vr_haptic_source_test tests/vr_haptic_source_test.cpp)
add_test(NAME vr_haptic_source COMMAND vr_haptic_source_test)
add_executable(vr_haptic_bridge_test tests/vr_haptic_bridge_test.cpp src/vr_rumble_bridge.cpp src/vr_custom_haptic_bridge.cpp)
target_link_libraries(vr_haptic_bridge_test PRIVATE tg_events)
add_test(NAME vr_haptic_bridge COMMAND vr_haptic_bridge_test)
add_executable(vr_rumble_filter_test tests/vr_rumble_filter_test.cpp)
target_link_libraries(vr_rumble_filter_test PRIVATE tg_events)
add_test(NAME vr_rumble_filter COMMAND vr_rumble_filter_test)
add_executable(vr_rumble_bridge_test tests/vr_rumble_bridge_test.cpp)
add_test(NAME vr_rumble_bridge COMMAND vr_rumble_bridge_test)
add_executable(vr_custom_haptic_state_test tests/vr_custom_haptic_state_test.cpp)
add_test(NAME vr_custom_haptic_state COMMAND vr_custom_haptic_state_test)
add_executable(vr_custom_haptic_bridge_test tests/vr_custom_haptic_bridge_test.cpp)
add_test(NAME vr_custom_haptic_bridge COMMAND vr_custom_haptic_bridge_test)
add_executable(transport_probe tests/transport_probe.cpp src/truegear_transport.cpp)
target_compile_definitions(transport_probe PRIVATE TrueGear_TRANSPORT_TEST_PORT=18764)
target_link_libraries(transport_probe PRIVATE winhttp crypt32)
add_executable(runtime_probe tests/runtime_probe.cpp src/runtime.cpp src/truegear_transport.cpp src/truegear_player.cpp)
target_compile_definitions(runtime_probe PRIVATE TrueGear_TRANSPORT_TEST_PORT=18765)
target_link_libraries(runtime_probe PRIVATE tg_events winhttp crypt32)
add_executable(sdk_player_test tests/sdk_player_test.cpp src/truegear_transport.cpp src/truegear_player.cpp)
target_compile_definitions(sdk_player_test PRIVATE TrueGear_TRANSPORT_TEST_PORT=18766)
target_link_libraries(sdk_player_test PRIVATE winhttp crypt32)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME transport_mock COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_transport_mock.py $<TARGET_FILE:transport_probe>)
add_test(NAME runtime_mock COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_runtime_mock.py $<TARGET_FILE:runtime_probe> ${CMAKE_CURRENT_SOURCE_DIR}/config/titanfall2-truegear.json)
add_test(NAME sdk_player_mock COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sdk_player_mock.py $<TARGET_FILE:sdk_player_test>)
set_tests_properties(transport_mock runtime_mock sdk_player_mock PROPERTIES TIMEOUT 30)
endif()
